{{- /* Fetches a YouTube RSS feed (channel or playlist) at build time. Returns (dict "title" "link" "videos"); "videos" is empty when the feed can't be read, so an offline build still succeeds. Call with partialCached, keyed by the feed URL. */ -}} {{- $url := . -}} {{- $result := dict "title" "" "link" "" "videos" slice -}} {{- with try (resources.GetRemote $url) -}} {{- with .Err -}} {{- warnf "youtube feed %s: %s" $url . -}} {{- else with .Value -}} {{- $feed := . | transform.Unmarshal -}} {{- $links := $feed.link | default slice -}} {{- if reflect.IsMap $links }}{{ $links = slice $links }}{{ end -}} {{- $link := "" -}} {{- range $links -}} {{- if eq (index . "-rel") "alternate" }}{{ $link = index . "-href" }}{{ end -}} {{- end -}} {{- $entries := $feed.entry | default slice -}} {{- if reflect.IsMap $entries }}{{ $entries = slice $entries }}{{ end -}} {{- $videos := slice -}} {{- range $entries -}} {{- $videos = $videos | append (dict "title" .title "url" (printf "https://www.youtube.com/watch?v=%s" .videoId) "thumb" (printf "https://i.ytimg.com/vi/%s/mqdefault.jpg" .videoId) "date" (time.AsTime .published) ) -}} {{- end -}} {{- $result = dict "title" $feed.title "link" $link "videos" $videos -}} {{- else -}} {{- warnf "youtube feed %s: not found" $url -}} {{- end -}} {{- end -}} {{- return $result -}}
